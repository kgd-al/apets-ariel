\tikzset{
 cpg label/.initial={},
 cpg radius/.initial=0.25,
 cpg/.pic={
  \pgfkeysgetvalue{/tikz/cpg radius}{\R}
  \pgfkeysgetvalue{/tikz/cpg label}{\label}
  \ifx\mylabel\empty\else
   \node[circle, draw=red, minimum size=2*\R cm, inner sep=0pt] (\label) at (0,0) {};
  \fi
  \begin{scope}
    \clip (0,0) circle (\R);
    \draw[smooth, samples=80, domain=-\R:\R]
      plot (\x, {.5*\R*0.6 * sin(\x / \R * 180)});
  \end{scope}
 },
 >=stealth'
}
