\documentclass[tikz,border=0pt]{standalone}
\usepackage{ifthen}
\begin{document}

% Which layer to render: all | cpg | abcpg | policy
% (override from the command line, see export.sh)
\providecommand{\layer}{all}
\newcommand{\showlayer}[1]{%
  \ifthenelse{\equal{\layer}{all}\OR\equal{\layer}{#1}}{\csname draw#1\endcsname}{}}

\def\w{3}
\def\h{1}
\def\ih{.25}
\tikzset{
 component/.style={font=\sffamily\bfseries},
 label/.style={font=\tiny, fill=white, inner sep=1pt}
}

\newcommand{\drawpolicy}{%
  \draw[fill=blue!15]
    (0,3*\h) -- ++(\w,0) -- ++(0,-\h+.5*\ih)
             -- ++(-.2*\w,0) -- ++(0,-\ih) -- ++(-.2*\w,0) -- ++(0, \ih)
             -- ++(-.2*\w,0) -- ++(0,-\ih) -- ++(-.2*\w,0) -- ++(0, \ih)
             -- ++(-.2*\w, 0) -- cycle;
  \node [component] at (.5*\w,2.5*\h) {Policy};
}

\newcommand{\drawcpg}{%
  \draw[fill=green!15]
    (0,0) -- ++(\w, 0) -- ++(0, \h-.5*\ih) -- ++(-.2*\w, 0) 
          -- ++(-.1*\w, \ih) -- ++(-.1*\w, -\ih)
          -- ++(-.1*\w, \ih) -- ++(-.1*\w, -\ih)
          -- ++(-.1*\w, \ih) -- ++(-.1*\w, -\ih)
          -- ++(-.2*\w, 0) -- cycle;
  \node [component] at (.5*\w,.5*\h) {CPG};}

\newcommand{\drawabcpg}{%
  \draw[fill=orange!25]
    (0,\h-.5*\ih) -- ++(.2*\w,0) 
          -- ++(.1*\w, \ih) -- ++(.1*\w, -\ih)
          -- ++(.1*\w, \ih) -- ++(.1*\w, -\ih)
          -- ++(.1*\w, \ih) -- ++(.1*\w, -\ih)
           -- ++(.2*\w,0) -- ++(0, \h+\ih)
          -- ++(-.2*\w,0) -- ++(0,-\ih) -- ++(-.2*\w,0) -- ++(0, \ih)
          -- ++(-.2*\w,0) -- ++(0,-\ih) -- ++(-.2*\w,0) -- ++(0, \ih)
           -- ++(-.2*\w,0) -- cycle;
  \node [component] at (.5*\w,1.5*\h) {$\alpha\beta $-CPG};}

\newcommand{\env}{
 \node at (1.5*\w,2*\h) (E) [draw, font=\small] {Environment};
}

\newcommand{\drawbase}{%
 \env
 \draw [->] (\w, .5*\h) -| (E) node [pos=.5, label] {Actions};
}

\newcommand{\drawloop}{%
 \env
 \draw [<-] (\w, 2.5*\h) -| (E) node [pos=.5, label] {Observations};
 \draw [->] (\w, 1.5*\h) -| (E) node [pos=.5, label] {Actions};
}

\newcommand{\canvasbox}{
 \def\m{.1}
 \useasboundingbox (-\m,-\m) rectangle (\m+2*\w, \m+3*\h);
}

\begin{tikzpicture}[line join=round, thick, >=stealth,]
  \canvasbox
  \showlayer{policy}
  \showlayer{cpg}
  \showlayer{abcpg}
  \ifthenelse{\equal{\layer}{base}}{
   \showlayer{base}
  }{}
  \showlayer{loop}
\end{tikzpicture}
\end{document}
